cmake_minimum_required(VERSION 3.15)
project(HtitNativeRegressions LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(RNS_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(RNS_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(RNS_BUILD_INTEROP OFF CACHE BOOL "" FORCE)
set(RNS_USE_FS OFF CACHE BOOL "" FORCE)
set(RNS_USE_PROVISIONING OFF CACHE BOOL "" FORCE)
set(RNS_PERSIST_PATHS OFF CACHE BOOL "" FORCE)
set(RNS_PERSIST_KNOWN_DESTINATIONS OFF CACHE BOOL "" FORCE)
set(RNS_PERSIST_HASHLIST OFF CACHE BOOL "" FORCE)
set(VENDOR ${CMAKE_CURRENT_SOURCE_DIR}/../../microreticulum/executor-libs)
add_subdirectory(${VENDOR}/microReticulum rns)
if(NOT RNS_MSGPACK_SOURCE_DIR)
  FetchContent_GetProperties(msgpack SOURCE_DIR RNS_MSGPACK_SOURCE_DIR)
endif()
find_package(BZip2 REQUIRED)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
execute_process(COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/../../microreticulum/executor/prepare_native.py msgpack ${RNS_MSGPACK_SOURCE_DIR} ${CMAKE_CURRENT_BINARY_DIR}/msgpack RESULT_VARIABLE prepare_status)
if(NOT prepare_status EQUAL 0)
  message(FATAL_ERROR "MessagePack preparation failed")
endif()
target_include_directories(microReticulum BEFORE PUBLIC ${CMAKE_CURRENT_BINARY_DIR}/msgpack)
target_link_libraries(microReticulum PUBLIC BZip2::BZip2)
add_executable(lxmf_recovery lxmf_recovery.cpp ${VENDOR}/microLXMF/src/LXMF/LXMessage.cpp ${VENDOR}/microLXMF/src/LXMF/LXStamper.cpp)
target_include_directories(lxmf_recovery PRIVATE ${VENDOR}/microLXMF/src/LXMF ${VENDOR}/microReticulum/src/microReticulum)
target_link_libraries(lxmf_recovery PRIVATE microReticulum)
enable_testing()
add_test(NAME lxmf_recovery COMMAND lxmf_recovery)
add_executable(msgpack_fixed_array msgpack_fixed_array.cpp)
target_link_libraries(msgpack_fixed_array PRIVATE microReticulum)
add_test(NAME msgpack_fixed_array COMMAND msgpack_fixed_array)
# LXMF/Type.h uses RNS constants before including the identically named RNS header.
target_compile_options(lxmf_recovery PRIVATE -include ${VENDOR}/microReticulum/src/microReticulum/Type.h)

set(HTIT_ROUTER_SOURCE "${VENDOR}/microLXMF/src/LXMF/LXMRouter.cpp" CACHE FILEPATH "Router implementation under test")
target_compile_definitions(lxmf_recovery PRIVATE HTIT_ROUTER_SOURCE="${HTIT_ROUTER_SOURCE}")
option(HTIT_CROSS_PEER "Exercise proof from a different peer than the stale registry entry" OFF)
if(HTIT_CROSS_PEER)
  target_compile_definitions(lxmf_recovery PRIVATE HTIT_PROBE_CROSS_PEER)
endif()
option(HTIT_LATE_CLOSE "Close an outbound link after destroying its router" OFF)
if(HTIT_LATE_CLOSE)
  target_compile_definitions(lxmf_recovery PRIVATE HTIT_PROBE_LATE_CLOSE)
endif()
set_tests_properties(lxmf_recovery PROPERTIES TIMEOUT 10)
# Optional negative check against an earlier housekeeping implementation.
set(HTIT_TRANSPORT_SOURCE "" CACHE FILEPATH "Override Transport.cpp for regression comparison")
if(HTIT_TRANSPORT_SOURCE)
  target_include_directories(microReticulum PRIVATE ${VENDOR}/microReticulum/src/microReticulum)
  get_target_property(rns_sources microReticulum SOURCES)
  list(FILTER rns_sources EXCLUDE REGEX "/Transport[.]cpp$")
  set_property(TARGET microReticulum PROPERTY SOURCES ${rns_sources} ${HTIT_TRANSPORT_SOURCE})
endif()

# The November parser and signed-proof reproductions use the same vendored code.
foreach(probe lxmf_nested_fields lxmf_signed_proof_uaf)
  add_executable(${probe} ../upstream/${probe}.cpp
    ${VENDOR}/microLXMF/src/LXMF/LXMessage.cpp
    ${VENDOR}/microLXMF/src/LXMF/LXStamper.cpp)
  target_include_directories(${probe} PRIVATE ${VENDOR}/microLXMF/src/LXMF ${VENDOR}/microReticulum/src/microReticulum)
  target_compile_options(${probe} PRIVATE -include ${VENDOR}/microReticulum/src/microReticulum/Type.h)
  target_compile_definitions(${probe} PRIVATE HTIT_ROUTER_SOURCE="${HTIT_ROUTER_SOURCE}")
  target_link_libraries(${probe} PRIVATE microReticulum)
endforeach()
add_test(NAME nested_fields COMMAND lxmf_nested_fields 4000)
add_test(NAME signed_proof_after_router_destroy COMMAND lxmf_signed_proof_uaf)
set_tests_properties(nested_fields signed_proof_after_router_destroy PROPERTIES TIMEOUT 30)
